# Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the MCP server, agent-core services, and API routes that let Claude analyze a real GitHub repo through real MCP tool calls and produce a structured plan visible in the UI.

**Architecture:** Five vertical PRs, each on its own branch. The MCP server runs as a child process (stdio transport). The agent loop is: Claude API ↔ MCPClientManager ↔ repo-agent-mcp-server ↔ local clone. State transitions are persisted to PostgreSQL via Prisma.

**Tech Stack:** `@modelcontextprotocol/sdk`, `@anthropic-ai/sdk`, `@octokit/rest`, `simple-git`, `execa`, Prisma 5, Fastify 4, Next.js 15, Vitest, TypeScript (NodeNext modules).

```typescript
const PATTERNS: RegExp[] = [
  /^[A-Z_]+=.*/gm,
  /sk-[a-zA-Z0-9]{32,}/g,
  /ghp_[a-zA-Z0-9]{36}/g,
  /-----BEGIN[^\n]*PRIVATE KEY-----/g,
  /Bearer [a-zA-Z0-9\-._~+/]+=*/g,
]

export function redact(input: string): string {
  let result = input
  for (const pattern of PATTERNS) {
    result = result.replace(pattern, '[REDACTED]')
  }
  return result
}
```

```typescript
import path from 'node:path'

export function validatePath(inputPath: string, repoRoot: string): string {
  const resolvedRoot = path.resolve(repoRoot)
  const resolved = path.resolve(repoRoot, inputPath)
  // Add path.sep to prevent /tmp/repo-attack from passing a /tmp/repo root check
  if (!resolved.startsWith(resolvedRoot + path.sep) && resolved !== resolvedRoot) {
    throw new Error('Path traversal attempt blocked.')
  }
  return resolved
}
```

```typescript
import crypto from 'node:crypto'

export class EncryptionService {
  private readonly key: Buffer

  constructor(hexKey: string) {
    if (hexKey.length !== 64) {
      throw new Error('TOKEN_ENCRYPTION_KEY must be a 64-character hex string (32 bytes)')
    }
    this.key = Buffer.from(hexKey, 'hex')
  }

  encrypt(plaintext: string): string {
    const iv = crypto.randomBytes(12)
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, iv)
    const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()])
    const authTag = cipher.getAuthTag()
    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted.toString('hex')}`
  }

  decrypt(ciphertext: string): string {
    const parts = ciphertext.split(':')
    if (parts.length !== 3) throw new Error('Invalid ciphertext format')
    const [ivHex, authTagHex, encryptedHex] = parts
    const iv = Buffer.from(ivHex, 'hex')
    const authTag = Buffer.from(authTagHex, 'hex')
    const encrypted = Buffer.from(encryptedHex, 'hex')
    const decipher = crypto.createDecipheriv('aes-256-gcm', this.key, iv)
    decipher.setAuthTag(authTag)
    return decipher.update(encrypted).toString('utf8') + decipher.final('utf8')
  }
}
```

```typescript
import path from 'node:path'

const BLOCKED_NAMES = new Set([
  '.npmrc', '.netrc',
  'id_rsa', 'id_ed25519', 'authorized_keys',
  'secrets.json', 'credentials.json',
])

const BLOCKED_PATTERNS = [
  /^\.env$/,
  /^\.env\./,
  /\.pem$/,
  /\.key$/,
  /\.p12$/,
  /\.pfx$/,
]

export function isBlocklisted(filePath: string): boolean {
  const base = path.basename(filePath)
  if (BLOCKED_NAMES.has(base)) return true
  return BLOCKED_PATTERNS.some(p => p.test(base))
}
```

```typescript
import fs from 'node:fs/promises'
import path from 'node:path'
import { validatePath } from '@repo-pilot/agent-core'

interface ListFilesArgs {
  directory: string
}

interface FileEntry {
  name: string
  type: 'file' | 'dir'
  path: string
}

export async function listFilesTool(
  args: ListFilesArgs,
  repoRoot: string
): Promise<{ content: [{ type: 'text'; text: string }] }> {
  const resolved = validatePath(args.directory, repoRoot)
  const entries = await fs.readdir(resolved, { withFileTypes: true })
  const result: FileEntry[] = entries.map(e => ({
    name: e.name,
    type: e.isDirectory() ? 'dir' : 'file',
    path: path.relative(repoRoot, path.join(resolved, e.name)),
  }))
  return { content: [{ type: 'text', text: JSON.stringify({ entries: result }) }] }
}
```

```typescript
import fs from 'node:fs/promises'
import { validatePath, redact } from '@repo-pilot/agent-core'
import { isBlocklisted } from '../utils/blocklist.js'

interface ReadFileArgs {
  path: string
}

export async function readFileTool(
  args: ReadFileArgs,
  repoRoot: string
): Promise<{ content: [{ type: 'text'; text: string }] }> {
  if (isBlocklisted(args.path)) {
    throw new Error('Access to this file is not permitted')
  }
  const resolved = validatePath(args.path, repoRoot)

  const content = redact(rawContent)
  const lineCount = content.split('\n').length
  const language = detectLanguage(args.path)
  return {
    content: [{ type: 'text', text: JSON.stringify({ content, lineCount, language }) }],
  }
}
```

```typescript
import { execa } from 'execa'

interface SearchRepoArgs {
  query: string
  fileGlob?: string
}

interface SearchMatch {
  file: string
  lineNumber: number
  lineContent: string
}

export async function searchRepoTool(
  args: SearchRepoArgs,
  repoRoot: string
): Promise<{ content: [{ type: 'text'; text: string }] }> {
  const rgArgs = ['--json', '--max-count', '50', args.query]
  if (args.fileGlob) rgArgs.push('--glob', args.fileGlob)

  // reject: false so we don't throw when rg exits 1 (no matches)
  const { stdout } = await execa('rg', rgArgs, { cwd: repoRoot, reject: false })

  const matches: SearchMatch[] = stdout
    .split('\n')
    .filter(Boolean)
    .map(line => { try { return JSON.parse(line) } catch { return null } })
    .filter((item): item is { type: string; data: { path: { text: string }; line_number: number; lines: { text: string } } } =>
      item?.type === 'match'
    )
    .map(item => ({
      file: item.data.path.text,
      lineNumber: item.data.line_number,
      lineContent: item.data.lines.text.trim(),
    }))

  return { content: [{ type: 'text', text: JSON.stringify({ matches }) }] }
}
```

```typescript
import { execa } from 'execa'

export async function getDiffTool(
  repoRoot: string
): Promise<{ content: [{ type: 'text'; text: string }] }> {
  const { stdout: diff } = await execa('git', ['diff', 'HEAD'], { cwd: repoRoot, reject: false })
  const { stdout: stat } = await execa('git', ['diff', 'HEAD', '--stat'], { cwd: repoRoot, reject: false })

  const statsMatch = stat.match(/(\d+) files? changed(?:, (\d+) insertions?\(\+\))?(?:, (\d+) deletions?\(-\))?/)
  const filesChanged = statsMatch ? parseInt(statsMatch[1], 10) : 0
  const insertions = statsMatch?.[2] ? parseInt(statsMatch[2], 10) : 0
  const deletions = statsMatch?.[3] ? parseInt(statsMatch[3], 10) : 0

  return {
    content: [{ type: 'text', text: JSON.stringify({ diff, filesChanged, insertions, deletions }) }],
  }
}
```

```typescript
        },
        required: ['query'],
      },
    },
    {
      name: 'read_file',
      description: 'Read the content of a single file. Secret redaction is applied.',
      inputSchema: {
        type: 'object',

      inputSchema: { type: 'object', properties: {} },
    },
    {
      name: 'get_github_issue',
      description: 'Fetch a single GitHub issue by number.',
      inputSchema: {

        properties: { issueNumber: { type: 'number' } },
        required: ['issueNumber'],
      },
    },
```

```typescript
import Anthropic from '@anthropic-ai/sdk'
import { redact } from './secret-redactor.js'

type Tool = {
  name: string
  description: string
  inputSchema: Record<string, unknown>
}

type ToolHandler = (name: string, input: unknown) => Promise<string>

export class ClaudeService {
  private readonly client: Anthropic

  constructor(apiKey: string) {
    this.client = new Anthropic({ apiKey })
  }

  async sendWithTools(
    systemPrompt: string,
    tools: Tool[],
    messages: Anthropic.MessageParam[],
    toolHandler?: ToolHandler
  ): Promise<Anthropic.Message> {
    const anthropicTools: Anthropic.Tool[] = tools.map(t => ({
      name: t.name,
      description: t.description,
      input_schema: t.inputSchema as Anthropic.Tool['input_schema'],
    }))

    let currentMessages = [...messages]

    // Agentic loop: keep calling until no more tool_use blocks
    while (true) {
      const response = await this.client.messages.create({
        model: 'claude-sonnet-4-6',
        max_tokens: 4096,
        system: systemPrompt,
        tools: anthropicTools.length > 0 ? anthropicTools : undefined,
        messages: currentMessages,
      })

      console.log(`[ClaudeService] tokens used: in=${response.usage.input_tokens} out=${response.usage.output_tokens}`)

      if (response.stop_reason !== 'tool_use' || !toolHandler) {
        return response
      }

      // Process all tool_use blocks in this response
      const toolUseBlocks = response.content.filter(
        (b): b is Anthropic.ToolUseBlock => b.type === 'tool_use'
      )

      const toolResults: Anthropic.ToolResultBlockParam[] = await Promise.all(
        toolUseBlocks.map(async block => {
          const rawResult = await toolHandler(block.name, block.input)
          const redacted = redact(rawResult)
          return {
            type: 'tool_result' as const,
            tool_use_id: block.id,
            content: redacted,
          }
        })
      )

      currentMessages = [
        ...currentMessages,
        { role: 'assistant', content: response.content },
        { role: 'user', content: toolResults },
      ]
    }
  }
}
```

```typescript
import { Client } from '@modelcontextprotocol/sdk/client/index.js'
import { StdioClientTransport } from '@modelcontextprotocol/sdk/client/stdio.js'

export class MCPClientManager {
  private client: Client | null = null
  private transport: StdioClientTransport | null = null

  constructor(
    private readonly serverPath: string,
    private readonly repoRoot: string,
    private readonly githubToken: string = '',
    private readonly githubOwner: string = '',
    private readonly githubRepo: string = ''
  ) {}

  async connect(): Promise<void> {
    this.transport = new StdioClientTransport({
      command: 'node',
      args: [this.serverPath],
      env: {
        ...process.env,
        REPO_ROOT: this.repoRoot,
        GITHUB_TOKEN: this.githubToken,
        GITHUB_OWNER: this.githubOwner,
        GITHUB_REPO: this.githubRepo,
      },
    })
    this.client = new Client(
      { name: 'repo-pilot-api', version: '0.1.0' },
      { capabilities: {} }
    )
    await this.client.connect(this.transport)
  }

  async listTools(): Promise<Array<{ name: string; description: string; inputSchema: unknown }>> {
    if (!this.client) throw new Error('MCPClientManager not connected')
    const { tools } = await this.client.listTools()
    return tools
  }

  async callTool(name: string, args: Record<string, unknown>): Promise<string> {
    if (!this.client) throw new Error('MCPClientManager not connected')
    const result = await this.client.callTool({ name, arguments: args })
    const textBlock = result.content.find(
      (b): b is { type: 'text'; text: string } => b.type === 'text'
    )
    return textBlock?.text ?? ''
  }

  async teardown(): Promise<void> {
    await this.client?.close()
    this.client = null
    this.transport = null
  }
}
```

```typescript
import type { AgentState } from '@repo-pilot/shared'

type PersistFn = (runId: string, state: AgentState, description: string) => Promise<void>

export class AgentStateMachine {
  private _currentState: AgentState = 'idle'
  private _errorMessage: string | null = null

  constructor(
    private readonly runId: string,
    private readonly persist: PersistFn
  ) {}

  get currentState(): AgentState {
    return this._currentState
  }

  get errorMessage(): string | null {
    return this._errorMessage
  }

  async transition(nextState: AgentState, description: string): Promise<void> {
    this._currentState = nextState
    await this.persist(this.runId, nextState, description)
  }

  async fail(errorMessage: string): Promise<void> {
    this._errorMessage = errorMessage
    await this.transition('failed', `Failed: ${errorMessage}`)
  }
}
```

```typescript
export interface RepoMeta {
  id: number
  owner: string
  name: string
  cloneUrl: string
}

export interface Issue {
  number: number
  title: string
  body: string
  labels: string[]
  state: string
}

export class GitHubService {
  private readonly octokit: Octokit

  constructor(private readonly token: string) {
    this.octokit = new Octokit({ auth: token })
  }

  async validateRepo(owner: string, repo: string): Promise<RepoMeta> {
    const { data } = await this.octokit.rest.repos.get({ owner, repo })
    return {
      id: data.id,
      owner: data.owner.login,
      name: data.name,
      cloneUrl: data.clone_url,
    }
  }

  async cloneRepo(cloneUrl: string, destPath: string): Promise<void> {
    // Inject token into URL for authenticated clone
    const authedUrl = cloneUrl.replace('https://', `https://x-access-token:${this.token}@`)
    const git = simpleGit()
    await git.clone(authedUrl, destPath)
  }

  async fetchIssues(owner: string, repo: string): Promise<Issue[]> {
    const { data } = await this.octokit.rest.issues.listForRepo({
      owner,
      repo,
      state: 'open',
      per_page: 30,
    })
    return data.map(issue => ({
      number: issue.number,
      title: issue.title,
      body: issue.body ?? '',
      labels: issue.labels.map(l => (typeof l === 'string' ? l : l.name ?? '')),
      state: issue.state,
    }))
  }
}
```

```typescript
import path from 'node:path'
import { AgentStateMachine, ClaudeService, MCPClientManager } from '@repo-pilot/agent-core'
import { db } from '../db.js'
import { parseEnv } from '../env.js'

const env = parseEnv()

// Path to the built MCP server binary
const MCP_SERVER_PATH = path.resolve('../../packages/mcp-server/dist/index.js')

export class AgentOrchestrator {
  async start(runId: string): Promise<void> {
    const run = await db.agentRun.findUniqueOrThrow({ where: { id: runId }, include: { repo: true } })

    const persist = async (rId: string, state: string, description: string) => {
      await db.agentRun.update({ where: { id: rId }, data: { currentState: state, status: 'running' } })
      await db.agentStep.create({
        data: {
          runId: rId,
          stepNumber: await db.agentStep.count({ where: { runId: rId } }) + 1,
          stepType: state,
          description,
          status: 'running',
        },
      })
    }

    const sm = new AgentStateMachine(runId, persist as Parameters<typeof AgentStateMachine>[1])
    const mcp = new MCPClientManager(
      MCP_SERVER_PATH,
      run.repo.localClonePath ?? '',
      '', // GitHub token handled separately in get_github_issue
      run.repo.owner,
      run.repo.name
    )

    try {
      await sm.transition('analyzing_repo', 'Starting repo analysis')
      await mcp.connect()

      const tools = await mcp.listTools()
      const claudeTools = tools.map(t => ({
        name: t.name,
        description: t.description ?? '',
        inputSchema: (t.inputSchema ?? { type: 'object', properties: {} }) as Record<string, unknown>,
      }))

      const claude = new ClaudeService(env.ANTHROPIC_API_KEY)

      const SYSTEM_PROMPT = `You are RepoPilot, an AI agent that helps developers automate codebase tasks safely.
Inspect the repository, then produce a structured JSON plan. Format:
{ "steps": [{ "stepName": string, "filesTouched": string[], "reason": string }] }
Do not propose any edits yet.`

      const ANALYZE_PROMPT = `Task: ${run.taskDescription}

Begin by calling list_files on the root directory, then search_repo for relevant files, then read the most relevant ones. After reading, output a JSON plan.`

      await sm.transition('planning', 'Claude analyzing repo')

      const response = await claude.sendWithTools(
        SYSTEM_PROMPT,
        claudeTools,
        [{ role: 'user', content: ANALYZE_PROMPT }],
        async (toolName, toolInput) => {
          return mcp.callTool(toolName, toolInput as Record<string, unknown>)
        }
      )

      // Extract plan JSON from Claude response
      const textBlock = response.content.find(b => b.type === 'text')
      let planJson: unknown = null
      if (textBlock && textBlock.type === 'text') {
        try {
          const match = textBlock.text.match(/\{[\s\S]*\}/)
          planJson = match ? JSON.parse(match[0]) : { raw: textBlock.text }
        } catch {
          planJson = { raw: textBlock.text }
        }
      }

      await db.agentRun.update({ where: { id: runId }, data: { planJson } })
      await sm.transition('waiting_for_plan_approval', 'Plan ready for review')

      // Mark all previous steps completed
      await db.agentStep.updateMany({
        where: { runId, status: 'running' },
        data: { status: 'completed', completedAt: new Date() },
      })
      await db.agentRun.update({ where: { id: runId }, data: { status: 'waiting' } })
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err)
      await sm.fail(message)
      await db.agentRun.update({ where: { id: runId }, data: { status: 'failed' } })
      await db.agentStep.updateMany({
        where: { runId, status: 'running' },
        data: { status: 'failed', errorMessage: message, completedAt: new Date() },
      })
    } finally {
      await mcp.teardown()
    }
  }
}
```

```typescript
import type { FastifyInstance } from 'fastify'
import path from 'node:path'
import { db } from '../db.js'
import { GitHubService } from '../services/github-service.js'
import { EncryptionService } from '@repo-pilot/agent-core'
import { parseEnv } from '../env.js'

const env = parseEnv()
const encryption = new EncryptionService(env.TOKEN_ENCRYPTION_KEY)

export async function reposRoute(app: FastifyInstance) {
  app.get('/repos', async () => {
    const repos = await db.repository.findMany()
    return repos.map(r => ({
      id: r.id,
      owner: r.owner,
      name: r.name,
      cloneUrl: r.cloneUrl,
      cloneStatus: r.cloneStatus,
      lastSyncedAt: r.lastSyncedAt?.toISOString() ?? null,
    }))
  })

  app.post<{ Body: { repoUrl: string; patToken: string } }>('/repos/connect', async (request, reply) => {
    const { repoUrl, patToken } = request.body

    // Parse owner/repo from URL
    const match = repoUrl.match(/github\.com\/([^/]+)\/([^/]+?)(?:\.git)?$/)
    if (!match) return reply.code(400).send({ error: 'Invalid GitHub repo URL' })
    const [, owner, name] = match

    const github = new GitHubService(patToken)
    let meta: Awaited<ReturnType<typeof github.validateRepo>>
    try {
      meta = await github.validateRepo(owner, name)
    } catch {
      return reply.code(400).send({ error: 'Invalid PAT or repository not found' })
    }

    const encryptedToken = encryption.encrypt(patToken)
    // Use a placeholder userId for Phase 1 (single-user MVP)
    const DEMO_USER_ID = 'demo-user'

    const repo = await db.repository.create({
      data: {
        userId: DEMO_USER_ID,
        githubRepoId: meta.id,
        owner: meta.owner,
        name: meta.name,
        cloneUrl: meta.cloneUrl,
        encryptedToken,
        cloneStatus: 'pending',
      },
    })

    const destPath = path.join(env.REPO_ROOT, repo.id)

    // Fire-and-forget clone (no queue — single-user MVP)
    github.cloneRepo(meta.cloneUrl, destPath)
      .then(() => db.repository.update({ where: { id: repo.id }, data: { cloneStatus: 'ready', localClonePath: destPath, lastSyncedAt: new Date() } }))
      .catch(() => db.repository.update({ where: { id: repo.id }, data: { cloneStatus: 'error' } }))

    return { repoId: repo.id, status: 'cloning' }
  })
}
```

```typescript
'use client'

import { useState, useEffect, useRef } from 'react'
import type { AgentRunResponse } from '@repo-pilot/shared'
import { fetchRun } from '../lib/api-client.js'

export function useAgentRun(runId: string | null) {
  const [run, setRun] = useState<AgentRunResponse | null>(null)
  const [error, setError] = useState<string | null>(null)
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)

  useEffect(() => {
    if (!runId) return

    const poll = async () => {
      try {
        const data = await fetchRun(runId)
        setRun(data)
        // Stop polling when run reaches a terminal state
        if (['complete', 'failed', 'waiting_for_plan_approval'].includes(data.currentState)) {
          if (intervalRef.current) clearInterval(intervalRef.current)
        }
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Unknown error')
      }
    }

    poll()
    intervalRef.current = setInterval(poll, 2000)
    return () => { if (intervalRef.current) clearInterval(intervalRef.current) }
  }, [runId])

  return { run, error }
}
```
